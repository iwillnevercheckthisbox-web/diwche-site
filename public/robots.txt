# Nothing here is blocked from search on purpose.
#
# /read, /bio and /r are meant to stay out of search results, and they say so
# themselves with `noindex` in their own markup. That is the directive that
# actually removes a page. A Disallow does the opposite of what it looks like:
# it stops the crawler fetching the page, so the crawler never reads the
# noindex, and a URL that picks up a single inbound link can still be listed —
# without a description, because nothing was allowed to read it.
#
# So the rule is: noindex to remove a page, Disallow only to spare the server.

User-agent: *
Allow: /

# The AI crawlers are a different question, because `noindex` means nothing to
# them. It is a search directive: it keeps a page out of a results list, and an
# assistant reading a page to answer a question is not building a results list.
# For these agents the fetch itself is the thing to allow or refuse, so here the
# Disallow really is the right tool — the one place on this site where it is.
#
# They are welcome on the marketing pages. Being the source an assistant quotes
# is worth more than the copy is worth withholding, and there is no moat in a
# page written to be read.
#
# They are not welcome on the three pages that carry someone else's material:
#   /read  — a funnel that renders figures about the account that walked it
#   /@…    — an account's own bio page (nginx maps /@handle onto /bio)
#   /r/…   — a shared report, containing captions a client wrote
# None of that is secret, and none of it is ours to feed to a model.
#
# Each agent below gets its own group, so it stops falling under `*` and reads
# only these lines. Everything not named here stays allowed.

User-agent: ClaudeBot
User-agent: Claude-User
User-agent: Claude-SearchBot
User-agent: GPTBot

User-agent: ChatGPT-User
User-agent: Google-Extended
User-agent: Applebot-Extended
User-agent: meta-externalagent
User-agent: PerplexityBot
User-agent: Perplexity-User
Disallow: /read
Disallow: /fa/read
Disallow: /de/read
Disallow: /bio

Disallow: /@
